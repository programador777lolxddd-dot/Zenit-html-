Condicionales y bucles

    
if edad >= 18 {
  imprimir("Mayor de edad")
} else if edad >= 13 {
  imprimir("Adolescente")
} else {
  imprimir("Menor de edad")
}

while n > 0 {
  n = n - 1
}

for i in 1..10 {
  imprimir(i)
}

for fruta in ["manzana", "pera"] {
  imprimir(fruta)
}


    
Funciones

    
func saludar(nombre) {
  return "¡Hola, " + nombre + "!"
}

imprimir(saludar("mundo"))

# Función anónima
var doble = func(x) { return x * 2 }
imprimir(doble(21))


    
Listas

    
var nums = [1, 2, 3]
agregar(nums, 4)
imprimir(longitud(nums))   # 4
imprimir(nums[0])          # 1
nums[0] = 99
for n in nums { imprimir(n) }


    
Funciones integradas

    

      
FunciónDescripción

      
imprimir(...) / print(...)Muestra valores en la salida

      
longitud(x) / len(x)Longitud de texto o lista

      
tipo(x)"numero" | "texto" | "lista" | "booleano" | "nulo" | "funcion"

      
numero(x) / texto(x)Conversión de tipos

      
rango(a, b)Igual que a..b

      
agregar(lista, v) / quitar(lista)Añade / elimina el último elemento

      
unir(lista, sep) / dividir(texto, sep)Une / separa

      
mayusculas(s) / minusculas(s)Cambia el caso

      
contiene(x, y)¿Contiene?

      
absoluto, raiz, potencia, redondear, piso, techo, aleatorio, minimo, maximoMatemáticas

      
error(msg)Lanza un error controlado
